#1.1
SELECT p.Name
FROM Patrons AS p
JOIN CheckedOut AS co ON co.CardNum = p.CardNum
GROUP BY p.CardNum, p.Name
ORDER BY COUNT(*) DESC, p.Name
LIMIT 1;

#1.2
SELECT t.Author
FROM Titles AS t
GROUP BY t.Author
HAVING COUNT(DISTINCT t.ISBN) > 1;

#1.3
SELECT t.Author
FROM Titles AS t
JOIN Inventory AS i ON i.ISBN = t.ISBN
GROUP BY t.Author
HAVING COUNT(*) > 1;

#1.4
SELECT
  p.Name AS Name,
  COUNT(co.Serial) AS Numbooks,
  CASE
    WHEN COUNT(co.Serial) > 2 THEN 'Platinum'
    WHEN COUNT(co.Serial) = 2 THEN 'Gold'
    WHEN COUNT(co.Serial) = 1 THEN 'Silver'
    ELSE 'Bronze'
  END AS Loyalty
FROM Patrons AS p
LEFT JOIN CheckedOut AS co ON co.CardNum = p.CardNum
GROUP BY p.CardNum, p.Name;

#2.4
SELECT DISTINCT p.Name
FROM Players AS p
JOIN (
  SELECT g.WhitePlayer AS pID
  FROM Games AS g
  JOIN Events AS e ON g.eID = e.eID
  WHERE e.Year = ...

  UNION

  SELECT g.BlackPlayer AS pID
  FROM Games AS g
  JOIN Events AS e ON g.eID = e.eID
  WHERE e.Year = ...
) AS PlayersYear ON PlayersYear.pID = p.pID;

FROM Events AS e
JOIN Games AS g ON e.eID = g.eID
JOIN Players AS p ON p.Name = "..."

   OR (g.BlackPlayer = p.pID AND g.Result = 'W');

#2.6
SELECT DISTINCT opp.Name
FROM Players AS target
JOIN Games AS g
  ON g.WhitePlayer = target.pID OR g.BlackPlayer = target.pID
JOIN Players AS opp
  ON (g.WhitePlayer = target.pID AND opp.pID = g.BlackPlayer)
  OR (g.BlackPlayer = target.pID AND opp.pID = g.WhitePlayer)
WHERE target.Name = "...";

#2.7
SELECT DISTINCT p.Name
FROM Players AS p
WHERE p.pID NOT IN (
  SELECT g.WhitePlayer
  FROM Games AS g
  WHERE g.Result = 'B'
)
AND p.pID NOT IN (
  SELECT g.BlackPlayer
  FROM Games AS g
  WHERE g.Result = 'W'
);

#3.2
INSERT IGNORE INTO CheckedOut (Serial, CardNum)
SELECT i.Serial, p.CardNum
FROM Patrons AS p
JOIN Inventory AS i
JOIN Titles AS t ON t.ISBN = i.ISBN
LEFT JOIN CheckedOut AS co ON co.Serial = i.Serial
WHERE p.Name = "...2"
  AND t.Title = "...1"
  AND co.Serial IS NULL
ORDER BY i.Serial, p.CardNum
LIMIT 1;
